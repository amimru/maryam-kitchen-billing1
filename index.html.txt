<script>
  const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxEdEJlskghsMKc7kvuO2DpClaDNU4lZoqzI-yavo9dJIaobSk9KYWjvF69FS7FC89UGQ/exec";

  function updateStatus() {
    const statusEl = document.getElementById('status');
    if (navigator.onLine) {
      statusEl.className = "status-bar online no-print";
      statusEl.innerText = "Online — Cloud Sync Enabled";
      syncOfflineInvoices();
    } else {
      statusEl.className = "status-bar offline no-print";
      statusEl.innerText = "Offline Mode — Data saved locally on mobile/PC";
    }
  }

  window.addEventListener('online', updateStatus);
  window.addEventListener('offline', updateStatus);
  updateStatus();

  function calculateTotal() {
    const qty = parseFloat(document.getElementById('qty').value) || 0;
    const rate = parseFloat(document.getElementById('rate').value) || 0;
    document.getElementById('netTotal').value = (qty * rate).toFixed(2);
  }

  function saveAndPrintInvoice() {
    const invoice = {
      invoiceNo: document.getElementById('invoiceNo').value,
      invoiceDate: document.getElementById('invoiceDate').value,
      customerName: document.getElementById('customerName').value,
      customerContact: document.getElementById('customerContact').value,
      itemName: document.getElementById('itemName').value,
      qty: document.getElementById('qty').value,
      rate: document.getElementById('rate').value,
      netTotal: document.getElementById('netTotal').value,
      paymentMethod: document.getElementById('paymentMethod').value,
      salesperson: "AMI"
    };

    if (!invoice.invoiceNo || !invoice.customerName) {
      alert("Please fill in the required fields.");
      return;
    }

    if (navigator.onLine) {
      sendToGoogleDrive(invoice);
    } else {
      let pending = JSON.parse(localStorage.getItem('pendingInvoices') || "[]");
      pending.push(invoice);
      localStorage.setItem('pendingInvoices', JSON.stringify(pending));
      alert("Invoice saved offline. It will sync to Google Sheets when internet connects.");
    }

    window.print();
  }

  function sendToGoogleDrive(data) {
    fetch(GOOGLE_SCRIPT_URL, {
      method: "POST",
      mode: "no-cors",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(data)
    }).then(() => console.log("Synced with Google Drive"))
      .catch(err => console.error("Sync failed:", err));
  }

  function syncOfflineInvoices() {
    let pending = JSON.parse(localStorage.getItem('pendingInvoices') || "[]");
    if (pending.length > 0) {
      pending.forEach(inv => sendToGoogleDrive(inv));
      localStorage.removeItem('pendingInvoices');
      console.log("All offline invoices synced to Google Drive.");
    }
  }
